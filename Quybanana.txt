-- fly toggle
local function makeButton(y, text)
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(1,-16,0,36)
    b.Position = UDim2.new(0,8,0,y)
    b.Text = text
    b.BackgroundColor3 = Color3.fromRGB(240,240,240)
    b.TextColor3 = Color3.fromRGB(20,20,20)
    b.Font = Enum.Font.Gotham
    b.TextSize = 16
    b.Parent = mainFrame
    return b
end

local flyBtn = makeButton(56, "Toggle Fly")
local invisBtn = makeButton(100, "Toggle Invisibility")
local speedUpBtn = makeButton(144, "Speed +10")
local speedDownBtn = makeButton(188, "Speed -10")
local closeBtn = makeButton(232, "Close Menu")

-- local state
local isFlying = false
local isInvisible = false
local currentSpeed = 16

flyBtn.MouseButton1Click:Connect(function()
    isFlying = not isFlying
    remote:FireServer("ToggleFly", isFlying)
    flyBtn.Text = isFlying and "Disable Fly" or "Toggle Fly"
end)

invisBtn.MouseButton1Click:Connect(function()
    isInvisible = not isInvisible
    remote:FireServer("ToggleInvis", isInvisible)
    invisBtn.Text = isInvisible and "Make Visible" or "Toggle Invisibility"
end)

speedUpBtn.MouseButton1Click:Connect(function()
    currentSpeed = currentSpeed + 10
    remote:FireServer("SetSpeed", currentSpeed)
end)
speedDownBtn.MouseButton1Click:Connect(function()
    currentSpeed = math.max(0, currentSpeed - 10)
    remote:FireServer("SetSpeed", currentSpeed)
end)

closeBtn.MouseButton1Click:Connect(function()
    screenGui.Enabled = false
end)

-- Short key to toggle UI (Right Ctrl)
ContextActionService:BindAction("ToggleQuyMenu", function(_, state, input)
    if state == Enum.UserInputState.Begin then
        screenGui.Enabled = not screenGui.Enabled
    end
end, false, Enum.KeyCode.RightControl)
